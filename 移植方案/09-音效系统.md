# 09 — 音效系统

## 概述

原游戏所有音效均通过 Web Audio API 实时合成（无音频文件）。MAKER 引擎使用 `SoundSource` 组件播放 .ogg 文件。

**移植策略**: 预生成所有音效为 .ogg 文件，放入 `assets/Sounds/` 目录。

---

## 音效清单与合成参数

以下列出所有需要生成的音效及其原始合成参数，用于指导 .ogg 文件的制作。

### 移动类音效

| 文件名 | 原始合成方式 | 时长 | 参数 |
|--------|-------------|------|------|
| `sfx_footstep.ogg` | Sine 频率扫描 150→30 Hz | 80ms | gain: 0.3, exponential ramp |
| `sfx_jump.ogg` | Square 波 200→30 Hz | 100ms | gain: 0.4, exponential ramp |
| `sfx_double_jump.ogg` | Noise burst | 80ms | bandpass 800Hz, gain: 0.3 |
| `sfx_dash.ogg` | Noise + lowpass 8000→300 Hz | 150ms | gain: 0.5, filter sweep |
| `sfx_climb.ogg` | Sawtooth 120→500 Hz | 100ms | gain: 0.2, linear ramp |
| `sfx_rail_on.ogg` | Triangle 800→1200 Hz | 200ms | gain: 0.3, linear ramp |
| `sfx_bounce.ogg` | Sine 60→200 Hz | 250ms | gain: 0.5, linear ramp |
| `sfx_land.ogg` | Sine 80→20 Hz | 50ms | gain: 0.3, quick decay |

### 战斗类音效

| 文件名 | 原始合成方式 | 时长 | 参数 |
|--------|-------------|------|------|
| `sfx_shotgun.ogg` | Sine kick 200→50 Hz + noise click | 120ms | kick gain: 0.6, noise gain: 0.4, noise duration: 40ms |
| `sfx_stream_shot.ogg` | Short pulse 1000→200 Hz | 40ms | gain: 0.2, sine wave |
| `sfx_needle.ogg` | FM synth 400 Hz + mod 800 Hz | 200ms | modIndex: 4, gain: 0.5, linear decay |
| `sfx_charge_tick.ogg` | Sine ping 1200 Hz | 50ms | gain: 0.3, sharp attack/decay |
| `sfx_ultimate.ogg` | Noise sweep + sine boom 30 Hz | 500ms | noise 0-200ms, boom 200-500ms, gain: 0.8 |
| `sfx_hit_env.ogg` | FM grain 300 Hz | 80ms | modIndex: 2, gain: 0.3, fast decay |
| `sfx_hit_enemy.ogg` | FM grain 500 Hz + resonance | 100ms | modIndex: 6, gain: 0.5, slight reverb tail |
| `sfx_death_explode.ogg` | Noise + sub 40 Hz | 300ms | noise gain: 0.6, sub gain: 0.4 |

### UI 类音效

| 文件名 | 原始合成方式 | 时长 | 参数 |
|--------|-------------|------|------|
| `sfx_menu_hover.ogg` | Sine 600 Hz | 30ms | gain: 0.15, gentle attack |
| `sfx_menu_click.ogg` | Sine 800→400 Hz | 60ms | gain: 0.3, sharp attack |
| `sfx_menu_transition.ogg` | Noise filtered sweep | 300ms | bandpass 200→2000 Hz, gain: 0.4 |

## 音效生成工具（Python 脚本）

用于从上述参数自动生成 .ogg 文件：

```python
# tools/generate_sounds.py
# 运行: pip install numpy scipy soundfile
import numpy as np
import soundfile as sf
import os

SAMPLE_RATE = 44100
OUTPUT_DIR = "assets/Sounds"

def sine_sweep(start_freq, end_freq, duration, gain=1.0):
    """频率扫描正弦波"""
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration))
    freq = np.exp(np.linspace(np.log(start_freq), np.log(end_freq), len(t)))
    phase = np.cumsum(freq / SAMPLE_RATE) * 2 * np.pi
    envelope = np.exp(-t / duration * 3)
    return np.sin(phase) * envelope * gain

def noise_burst(duration, gain=1.0, filter_freq=None):
    """噪声突发"""
    samples = int(SAMPLE_RATE * duration)
    noise = np.random.randn(samples)
    envelope = np.exp(-np.linspace(0, 5, samples))
    return noise * envelope * gain

def square_sweep(start_freq, end_freq, duration, gain=1.0):
    """方波频率扫描"""
    t = np.linspace(0, duration, int(SAMPLE_RATE * duration))
    freq = np.exp(np.linspace(np.log(start_freq), np.log(end_freq), len(t)))
    phase = np.cumsum(freq / SAMPLE_RATE) * 2 * np.pi
    envelope = np.exp(-t / duration * 3)
    return np.sign(np.sin(phase)) * envelope * gain

def save_ogg(filename, audio):
    """保存为 .ogg"""
    audio = np.clip(audio, -1, 1).astype(np.float32)
    filepath = os.path.join(OUTPUT_DIR, filename)
    sf.write(filepath, audio, SAMPLE_RATE, format='OGG')
    print(f"Generated: {filepath}")

# === 生成所有音效 ===

os.makedirs(OUTPUT_DIR, exist_ok=True)

# 移动类
save_ogg("sfx_footstep.ogg", sine_sweep(150, 30, 0.08, 0.3))
save_ogg("sfx_jump.ogg", square_sweep(200, 30, 0.1, 0.4))
save_ogg("sfx_double_jump.ogg", noise_burst(0.08, 0.3))
save_ogg("sfx_dash.ogg", noise_burst(0.15, 0.5))
save_ogg("sfx_climb.ogg", sine_sweep(120, 500, 0.1, 0.2))
save_ogg("sfx_rail_on.ogg", sine_sweep(800, 1200, 0.2, 0.3))
save_ogg("sfx_bounce.ogg", sine_sweep(60, 200, 0.25, 0.5))
save_ogg("sfx_land.ogg", sine_sweep(80, 20, 0.05, 0.3))

# 战斗类
shotgun = np.concatenate([
    sine_sweep(200, 50, 0.08, 0.6),
    noise_burst(0.04, 0.4)
])
save_ogg("sfx_shotgun.ogg", shotgun)
save_ogg("sfx_stream_shot.ogg", sine_sweep(1000, 200, 0.04, 0.2))
save_ogg("sfx_needle.ogg", sine_sweep(400, 400, 0.2, 0.5))  # 简化FM
save_ogg("sfx_charge_tick.ogg", sine_sweep(1200, 1200, 0.05, 0.3))
ultimate = np.concatenate([
    noise_burst(0.2, 0.6),
    sine_sweep(30, 30, 0.3, 0.8)
])
save_ogg("sfx_ultimate.ogg", ultimate)
save_ogg("sfx_hit_env.ogg", sine_sweep(300, 100, 0.08, 0.3))
save_ogg("sfx_hit_enemy.ogg", sine_sweep(500, 200, 0.1, 0.5))
save_ogg("sfx_death_explode.ogg", noise_burst(0.3, 0.6))

# UI 类
save_ogg("sfx_menu_hover.ogg", sine_sweep(600, 600, 0.03, 0.15))
save_ogg("sfx_menu_click.ogg", sine_sweep(800, 400, 0.06, 0.3))
save_ogg("sfx_menu_transition.ogg", noise_burst(0.3, 0.4))

print("All sounds generated!")
```

## MAKER 音频 API 集成

```lua
-- scripts/audio/AudioManager.lua
local AudioManager = {}

local scene_ = nil
local bgmNode_ = nil
local bgmSource_ = nil
local masterVolume_ = 0.8

function AudioManager.Init(scene)
    scene_ = scene
end

--- 播放 3D 音效
function AudioManager.Play3D(position, soundPath, volume)
    local node = scene_:CreateChild("SFX3D")
    node.position = position

    local source = node:CreateComponent("SoundSource3D")
    source.nearDistance = 1.0
    source.farDistance = 30.0
    source.gain = (volume or 1.0) * masterVolume_
    source.autoRemoveMode = REMOVE_NODE

    local sound = cache:GetResource("Sound", soundPath)
    if sound then
        source:Play(sound)
    end
end

--- 播放 2D 音效（不受位置影响）
function AudioManager.Play2D(soundPath, volume)
    local node = scene_:CreateChild("SFX2D")
    local source = node:CreateComponent("SoundSource")
    source.gain = (volume or 1.0) * masterVolume_
    source.autoRemoveMode = REMOVE_NODE
    source.soundType = "Effect"

    local sound = cache:GetResource("Sound", soundPath)
    if sound then
        source:Play(sound)
    end
end

--- 播放背景音乐
function AudioManager.PlayBGM(soundPath)
    if bgmNode_ then bgmNode_:Remove() end

    bgmNode_ = scene_:CreateChild("BGM")
    bgmSource_ = bgmNode_:CreateComponent("SoundSource")
    bgmSource_.soundType = "Music"
    bgmSource_.gain = 0.4 * masterVolume_

    local sound = cache:GetResource("Sound", soundPath)
    if sound then
        sound.looped = true
        bgmSource_:Play(sound)
    end
end

--- 设置主音量
function AudioManager.SetMasterVolume(vol)
    masterVolume_ = vol
    if bgmSource_ then
        bgmSource_.gain = 0.4 * masterVolume_
    end
end

return AudioManager
```

## 音效触发点集成

在各系统中调用 AudioManager：

```lua
-- 移动系统
PlayerMovement.OnJump = function(isDouble)
    local path = isDouble and "Sounds/sfx_double_jump.ogg" or "Sounds/sfx_jump.ogg"
    AudioManager.Play2D(path)
end

PlayerMovement.OnDash = function()
    AudioManager.Play2D("Sounds/sfx_dash.ogg")
end

PlayerMovement.OnLand = function()
    AudioManager.Play2D("Sounds/sfx_land.ogg", 0.5)
end

-- 战斗系统
WeaponSystem.OnShotgun = function()
    AudioManager.Play2D("Sounds/sfx_shotgun.ogg")
end

WeaponSystem.OnNeedle = function()
    AudioManager.Play2D("Sounds/sfx_needle.ogg")
end

WeaponSystem.OnUltimate = function()
    AudioManager.Play2D("Sounds/sfx_ultimate.ogg")
end

-- 命中
Juice.OnHitEnv = function(pos)
    AudioManager.Play3D(pos, "Sounds/sfx_hit_env.ogg")
end

Juice.OnHitEnemy = function(pos)
    AudioManager.Play3D(pos, "Sounds/sfx_hit_enemy.ogg")
end
```

## 脚步声系统

```lua
-- 脚步声计时器（每走一定距离播放）
local footstepTimer_ = 0
local FOOTSTEP_INTERVAL = 0.35  -- 秒

function UpdateFootsteps(dt, isMoving, isOnGround)
    if not isMoving or not isOnGround then
        footstepTimer_ = 0
        return
    end

    footstepTimer_ = footstepTimer_ + dt
    if footstepTimer_ >= FOOTSTEP_INTERVAL then
        footstepTimer_ = footstepTimer_ - FOOTSTEP_INTERVAL
        AudioManager.Play2D("Sounds/sfx_footstep.ogg", 0.3)
    end
end
```
