# 10 — 数据转换工具链

## 概述

将原始游戏的 JavaScript/JSON 数据转换为 MAKER 可用的 Lua 格式。需要在 Windows 上执行（目标机器）。

**前置依赖**: Python 3.10+, pip install numpy scipy soundfile

## 工具 1：convert_chunks.py

将 JSON chunk 数据转换为 Lua table 格式。

```python
#!/usr/bin/env python3
"""
convert_chunks.py
将 GAME/data/chunks/chunk_X_Z.json 转换为 Lua table 文件

用法: python convert_chunks.py <input_dir> <output_dir>
示例: python convert_chunks.py GAME/data/chunks assets/Data/chunks
"""

import json
import os
import sys
import glob

def color_int_to_rgb(color_int):
    """整数颜色转 0~1 浮点 RGB"""
    r = ((color_int >> 16) & 0xFF) / 255.0
    g = ((color_int >> 8) & 0xFF) / 255.0
    b = (color_int & 0xFF) / 255.0
    return round(r, 3), round(g, 3), round(b, 3)

def convert_chunk(input_path, output_path, negate_z=True):
    """转换单个 chunk 文件"""
    with open(input_path, 'r') as f:
        data = json.load(f)

    cx = data.get('cx', 0)
    cz = data.get('cz', 0)
    boxes = data.get('boxes', [])

    lines = []
    lines.append(f"-- Auto-generated from {os.path.basename(input_path)}")
    lines.append(f"-- Boxes: {len(boxes)}")
    lines.append("return {")
    lines.append(f"    cx = {cx}, cz = {cz},")
    lines.append("    boxes = {")

    for box in boxes:
        x = box['x']
        y = box['y']
        z = box['z']
        w = box.get('w', 1)
        h = box.get('h', 1)
        d = box.get('d', 1)
        color = box.get('color', 0xFFFFFF)

        # 坐标系转换: Three.js 右手系 → MAKER 左手系 (Z取反)
        if negate_z:
            z = -z

        r, g, b = color_int_to_rgb(color)
        lines.append(f"        {{{x},{y},{z},{w},{h},{d},{r},{g},{b}}},")

    lines.append("    }")
    lines.append("}")

    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write('\n'.join(lines))

    return len(boxes)

def main():
    if len(sys.argv) < 3:
        print("用法: python convert_chunks.py <input_dir> <output_dir>")
        sys.exit(1)

    input_dir = sys.argv[1]
    output_dir = sys.argv[2]

    json_files = glob.glob(os.path.join(input_dir, "chunk_*.json"))
    total_boxes = 0
    chunk_list = []

    print(f"找到 {len(json_files)} 个 chunk 文件")

    for json_path in sorted(json_files):
        filename = os.path.basename(json_path)
        lua_filename = filename.replace('.json', '.lua')
        output_path = os.path.join(output_dir, lua_filename)

        box_count = convert_chunk(json_path, output_path)
        total_boxes += box_count

        # 解析 cx, cz
        parts = filename.replace('chunk_', '').replace('.json', '').split('_')
        if len(parts) == 2:
            chunk_list.append({'cx': int(parts[0]), 'cz': int(parts[1])})

        print(f"  {filename} → {lua_filename} ({box_count} boxes)")

    # 生成 manifest
    manifest_path = os.path.join(output_dir, "manifest.lua")
    with open(manifest_path, 'w', encoding='utf-8') as f:
        f.write("-- Auto-generated chunk manifest\n")
        f.write(f"-- Total chunks: {len(chunk_list)}\n")
        f.write(f"-- Total boxes: {total_boxes}\n")
        f.write("return {\n")
        f.write("    chunks = {\n")
        for c in chunk_list:
            f.write(f"        {{cx={c['cx']}, cz={c['cz']}}},\n")
        f.write("    }\n")
        f.write("}\n")

    print(f"\n完成! 共转换 {len(json_files)} 个文件, {total_boxes} 个碰撞盒")
    print(f"Manifest: {manifest_path}")

if __name__ == '__main__':
    main()
```

## 工具 2：extract_rails.py

从 spawner.js 中提取轨道曲线控制点。

```python
#!/usr/bin/env python3
"""
extract_rails.py
从 GAME/scripts/world/spawner.js 提取轨道曲线数据

注意: 轨道数据可能嵌入在 spawnRail() 调用中，需要手动检查格式。
如果轨道是硬编码在 JS 中的，本脚本会提取 CatmullRomCurve3 的控制点。

用法: python extract_rails.py <spawner.js_path> <output_path>
"""

import re
import sys
import os

def extract_vector3_arrays(js_content):
    """从 JS 代码中提取 Vector3 数组"""
    # 匹配 new THREE.Vector3(x, y, z) 模式
    pattern = r'new\s+THREE\.Vector3\(\s*([-\d.]+)\s*,\s*([-\d.]+)\s*,\s*([-\d.]+)\s*\)'
    matches = re.findall(pattern, js_content)
    return [(float(x), float(y), float(z)) for x, y, z in matches]

def find_rail_definitions(js_content):
    """查找 spawnRail 调用中的控制点"""
    # 查找 spawnRail([...]) 调用
    # 模式: spawnRail([ ... ])
    rails = []

    # 简化方案：查找 CatmullRomCurve3 构造
    curve_pattern = r'CatmullRomCurve3\(\s*\[(.*?)\]\s*\)'
    curve_matches = re.findall(curve_pattern, js_content, re.DOTALL)

    for match in curve_matches:
        points = extract_vector3_arrays(match)
        if points:
            rails.append(points)

    return rails

def main():
    if len(sys.argv) < 3:
        print("用法: python extract_rails.py <spawner.js> <output.lua>")
        sys.exit(1)

    input_path = sys.argv[1]
    output_path = sys.argv[2]

    with open(input_path, 'r', encoding='utf-8') as f:
        content = f.read()

    rails = find_rail_definitions(content)

    if not rails:
        print("警告: 未找到轨道数据。可能需要手动提取。")
        print("请检查 spawner.js 中 spawnRail 的调用方式。")
        # 生成空文件
        rails = []

    # 生成 Lua
    lines = []
    lines.append("-- Auto-generated rail data")
    lines.append(f"-- Total rails: {len(rails)}")
    lines.append("-- 坐标已转换: Z 取反 (Three.js → MAKER)")
    lines.append("return {")

    for ri, rail in enumerate(rails):
        lines.append(f"    -- Rail {ri + 1} ({len(rail)} control points)")
        lines.append("    {")
        for x, y, z in rail:
            # Z 取反
            lines.append(f"        Vector3({x}, {y}, {-z}),")
        lines.append("    },")

    lines.append("}")

    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write('\n'.join(lines))

    print(f"提取了 {len(rails)} 条轨道 → {output_path}")

if __name__ == '__main__':
    main()
```

## 坐标系转换规则

| 轴 | Three.js | MAKER | 转换 |
|----|----------|-------|------|
| X | 右 | 右 | 不变 |
| Y | 上 | 上 | 不变 |
| Z | 屏幕外(后) | 前 | **取反** |

转换公式: `maker_z = -threejs_z`

应用到：
- 所有 chunk box 的 z 坐标
- 轨道控制点的 z 坐标
- 敌人生成位置（如果有硬编码数据）

## 注意事项

1. **文件大小**: 每个 chunk Lua 文件可能较大（7000+ 行），require 加载时会有短暂停顿，建议分帧加载。
2. **内存**: 72 万个碰撞盒在内存中约占 ~50MB（每个 7 个 float），需注意目标设备内存限制。
3. **编码**: 所有 Lua 文件使用 UTF-8 无 BOM。
4. **行尾**: Windows 上生成的文件使用 CRLF，MAKER 引擎应能处理两种格式。
5. **精度**: 坐标保留原始精度（float），颜色保留 3 位小数。
