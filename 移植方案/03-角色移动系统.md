# 03 — 角色移动系统

## 概述

原游戏实现了一套复杂的平台跳跃物理系统，完全自实现（无物理引擎依赖）。包含：基础移动、重力、二段跳、冲刺、贴墙滑行、攀爬、轨道滑行、气球弹跳。

MAKER 移植同样使用纯 Lua AABB 碰撞，不依赖 Bullet 物理。

---

## 实现：PlayerPhysics.lua

```lua
-- scripts/player/PlayerPhysics.lua
local Config = require("scripts/config")
local SpatialHash = require("scripts/world/SpatialHash")

local PlayerPhysics = {}

-- 玩家状态
local pos_ = Vector3.ZERO
local vel_ = Vector3.ZERO
local node_ = nil

-- 物理状态
local onGround_ = false
local onWall_ = false
local wallNormal_ = Vector3.ZERO
local jumpsLeft_ = 2
local hp_ = 100

-- 玩家碰撞体（AABB 半尺寸）
local HALF_W = Config.PLAYER_RADIUS   -- 0.35
local HALF_H = Config.PLAYER_HEIGHT / 2  -- 0.9
local HALF_D = Config.PLAYER_RADIUS   -- 0.35

function PlayerPhysics.Init(scene, startPos)
    pos_ = startPos:__clone()
    vel_ = Vector3.ZERO
    node_ = scene:CreateChild("Player")
    node_.position = pos_
    onGround_ = false
    jumpsLeft_ = Config.MAX_JUMPS
    hp_ = Config.MAX_HP
end

function PlayerPhysics.Update(dt)
    -- 重力
    if not onGround_ then
        vel_.y = vel_.y + Config.GRAVITY * dt
    end

    -- 摩擦（水平方向）
    local frictionCoeff = onGround_ and Config.FRICTION or Config.AIR_FRICTION
    vel_.x = vel_.x - vel_.x * frictionCoeff * dt
    vel_.z = vel_.z - vel_.z * frictionCoeff * dt

    -- 分轴碰撞检测与位移
    -- X 轴
    local newX = pos_.x + vel_.x * dt
    if not PlayerPhysics._checkCollisionAxis("x", newX, pos_.y, pos_.z) then
        pos_.x = newX
    else
        vel_.x = 0
        -- 检测是否贴墙
        onWall_ = not onGround_
        if onWall_ then
            wallNormal_ = PlayerPhysics._getWallNormal("x", newX, pos_.y, pos_.z)
        end
    end

    -- Z 轴
    local newZ = pos_.z + vel_.z * dt
    if not PlayerPhysics._checkCollisionAxis("z", pos_.x, pos_.y, newZ) then
        pos_.z = newZ
    else
        vel_.z = 0
        onWall_ = not onGround_
        if onWall_ then
            wallNormal_ = PlayerPhysics._getWallNormal("z", pos_.x, pos_.y, newZ)
        end
    end

    -- Y 轴
    local newY = pos_.y + vel_.y * dt
    local wasOnGround = onGround_
    onGround_ = false

    if vel_.y <= 0 then
        -- 下落：检测脚下碰撞
        if PlayerPhysics._checkCollisionAxis("y_down", pos_.x, newY, pos_.z) then
            -- 落地
            local groundY = PlayerPhysics._findGroundY(pos_.x, newY, pos_.z)
            pos_.y = groundY
            vel_.y = 0
            onGround_ = true
            onWall_ = false
            jumpsLeft_ = Config.MAX_JUMPS
        else
            pos_.y = newY
        end
    else
        -- 上升：检测头顶碰撞
        if PlayerPhysics._checkCollisionAxis("y_up", pos_.x, newY, pos_.z) then
            vel_.y = 0
            -- pos_.y 不变（卡住）
        else
            pos_.y = newY
        end
    end

    -- 贴墙滑行：限制下落速度
    if onWall_ and vel_.y < Config.WALL_SLIDE_SPEED then
        vel_.y = Config.WALL_SLIDE_SPEED
    end

    -- 更新节点位置
    node_.position = pos_
end

-- === 碰撞检测辅助 ===

function PlayerPhysics._checkCollisionAxis(axis, x, y, z)
    -- 计算玩家 AABB
    local minX, maxX, minY, maxY, minZ, maxZ

    if axis == "x" then
        minX = x - HALF_W; maxX = x + HALF_W
        minY = pos_.y; maxY = pos_.y + Config.PLAYER_HEIGHT
        minZ = pos_.z - HALF_D; maxZ = pos_.z + HALF_D
    elseif axis == "z" then
        minX = pos_.x - HALF_W; maxX = pos_.x + HALF_W
        minY = pos_.y; maxY = pos_.y + Config.PLAYER_HEIGHT
        minZ = z - HALF_D; maxZ = z + HALF_D
    elseif axis == "y_down" then
        minX = pos_.x - HALF_W; maxX = pos_.x + HALF_W
        minY = y; maxY = y + Config.PLAYER_HEIGHT
        minZ = pos_.z - HALF_D; maxZ = pos_.z + HALF_D
    elseif axis == "y_up" then
        minX = pos_.x - HALF_W; maxX = pos_.x + HALF_W
        minY = y; maxY = y + Config.PLAYER_HEIGHT
        minZ = pos_.z - HALF_D; maxZ = pos_.z + HALF_D
    end

    return SpatialHash.CheckCollision(minX, minY, minZ, maxX, maxY, maxZ)
end

function PlayerPhysics._findGroundY(x, y, z)
    -- 向下搜索最近的碰撞体顶面
    local colliders = SpatialHash.GetNearby(x, y, z)
    local bestY = y
    for _, col in ipairs(colliders) do
        if col.maxY > y and col.maxY < y + Config.PLAYER_HEIGHT then
            if x + HALF_W > col.minX and x - HALF_W < col.maxX and
               z + HALF_D > col.minZ and z - HALF_D < col.maxZ then
                if col.maxY > bestY then
                    bestY = col.maxY
                end
            end
        end
    end
    return bestY
end

function PlayerPhysics._getWallNormal(axis, x, y, z)
    if axis == "x" then
        return vel_.x > 0 and Vector3(-1, 0, 0) or Vector3(1, 0, 0)
    else
        return vel_.z > 0 and Vector3(0, 0, -1) or Vector3(0, 0, 1)
    end
end

-- === 公共接口 ===

function PlayerPhysics.GetPosition() return pos_ end
function PlayerPhysics.GetVelocity() return vel_ end
function PlayerPhysics.GetNode() return node_ end
function PlayerPhysics.IsOnGround() return onGround_ end
function PlayerPhysics.IsOnWall() return onWall_ end
function PlayerPhysics.GetWallNormal() return wallNormal_ end
function PlayerPhysics.GetHealth() return hp_ end
function PlayerPhysics.GetJumpsLeft() return jumpsLeft_ end

function PlayerPhysics.SetVelocity(v)
    vel_ = v:__clone()
end

function PlayerPhysics.AddVelocity(v)
    vel_ = vel_ + v
end

function PlayerPhysics.SetOnGround(val)
    onGround_ = val
    if val then jumpsLeft_ = Config.MAX_JUMPS end
end

function PlayerPhysics.ConsumeJump()
    if jumpsLeft_ > 0 then
        jumpsLeft_ = jumpsLeft_ - 1
        return true
    end
    return false
end

function PlayerPhysics.ResetJumps()
    jumpsLeft_ = Config.MAX_JUMPS
end

function PlayerPhysics.TakeDamage(amount)
    hp_ = math.max(0, hp_ - amount)
    return hp_ <= 0  -- 返回是否死亡
end

return PlayerPhysics
```

## 实现：PlayerMovement.lua（高层状态机）

```lua
-- scripts/player/PlayerMovement.lua
local Config = require("scripts/config")
local PlayerPhysics = require("scripts/player/PlayerPhysics")
local Mechanics = require("scripts/world/Mechanics")

local PlayerMovement = {}

-- 状态
local state_ = "normal"  -- "normal" | "dashing" | "climbing" | "onRail" | "wallSlide"
local dashTimer_ = 0
local dashDir_ = Vector3.ZERO

-- 冲刺冷却
local dashCooldown_ = 0
local DASH_CD = 0.3  -- 冲刺冷却时间

function PlayerMovement.Update(dt, cameraYaw)
    -- 冷却计时
    if dashCooldown_ > 0 then dashCooldown_ = dashCooldown_ - dt end

    if state_ == "dashing" then
        PlayerMovement._updateDash(dt)
        return
    end

    if state_ == "onRail" then
        Mechanics.UpdateRail(dt)
        -- 跳跃退出轨道
        if input:GetKeyPress(KEY_SPACE) then
            Mechanics.ExitRail()
            state_ = "normal"
            PlayerPhysics.SetVelocity(Vector3(0, Config.JUMP_FORCE * 0.7, 0))
        end
        return
    end

    if state_ == "climbing" then
        PlayerMovement._updateClimb(dt, cameraYaw)
        -- 跳跃退出攀爬
        if input:GetKeyPress(KEY_SPACE) then
            state_ = "normal"
            local wallN = PlayerPhysics.GetWallNormal()
            PlayerPhysics.SetVelocity(Vector3(
                wallN.x * 10,
                Config.JUMP_FORCE * 0.8,
                wallN.z * 10
            ))
            PlayerPhysics.ConsumeJump()
        end
        return
    end

    -- === 正常状态 ===

    -- 移动方向（相对相机偏航）
    local moveDir = PlayerMovement._getMoveDirection(cameraYaw)

    -- 应用水平速度
    if moveDir:Length() > 0.01 then
        local vel = PlayerPhysics.GetVelocity()
        vel.x = vel.x + moveDir.x * Config.MOVE_SPEED * dt * 10
        vel.z = vel.z + moveDir.z * Config.MOVE_SPEED * dt * 10

        -- 限制水平速度
        local hSpeed = math.sqrt(vel.x * vel.x + vel.z * vel.z)
        if hSpeed > Config.MOVE_SPEED then
            local ratio = Config.MOVE_SPEED / hSpeed
            vel.x = vel.x * ratio
            vel.z = vel.z * ratio
        end
        PlayerPhysics.SetVelocity(vel)
    end

    -- 跳跃
    if input:GetKeyPress(KEY_SPACE) then
        if PlayerPhysics.ConsumeJump() then
            local vel = PlayerPhysics.GetVelocity()
            vel.y = Config.JUMP_FORCE
            PlayerPhysics.SetVelocity(vel)
            PlayerPhysics.SetOnGround(false)
            -- TODO: 播放跳跃音效
        end
    end

    -- 冲刺
    if input:GetKeyPress(KEY_SHIFT) and dashCooldown_ <= 0 then
        PlayerMovement._startDash(cameraYaw)
    end

    -- 攀爬检测（贴墙时按 W）
    if PlayerPhysics.IsOnWall() and input:GetKeyDown(KEY_W) and not PlayerPhysics.IsOnGround() then
        state_ = "climbing"
    end

    -- 贴墙滑行（贴墙时自动）
    if PlayerPhysics.IsOnWall() and not PlayerPhysics.IsOnGround() and state_ ~= "climbing" then
        state_ = "wallSlide"
        -- 贴墙跳（按空格从墙上弹开）
        if input:GetKeyPress(KEY_SPACE) then
            local wallN = PlayerPhysics.GetWallNormal()
            PlayerPhysics.SetVelocity(Vector3(
                wallN.x * 12,
                Config.JUMP_FORCE,
                wallN.z * 12
            ))
            PlayerPhysics.ConsumeJump()
            state_ = "normal"
        end
    elseif state_ == "wallSlide" and (PlayerPhysics.IsOnGround() or not PlayerPhysics.IsOnWall()) then
        state_ = "normal"
    end

    -- 轨道检测
    local railResult = Mechanics.CheckRailProximity(PlayerPhysics.GetPosition())
    if railResult then
        state_ = "onRail"
        Mechanics.EnterRail(railResult.curveIndex, railResult.t)
    end

    -- 气球碰撞检测
    if Mechanics.CheckBalloonCollision(PlayerPhysics.GetPosition()) then
        local vel = PlayerPhysics.GetVelocity()
        vel.y = Config.BALLOON_BOUNCE_SPEED
        PlayerPhysics.SetVelocity(vel)
        PlayerPhysics.ResetJumps()
        -- TODO: 播放弹跳音效 + 粒子
    end
end

-- === 内部方法 ===

function PlayerMovement._getMoveDirection(cameraYaw)
    local dir = Vector3.ZERO
    local yawRad = math.rad(cameraYaw)
    local forward = Vector3(math.sin(yawRad), 0, math.cos(yawRad))
    local right = Vector3(math.cos(yawRad), 0, -math.sin(yawRad))

    if input:GetKeyDown(KEY_W) then dir = dir + forward end
    if input:GetKeyDown(KEY_S) then dir = dir - forward end
    if input:GetKeyDown(KEY_D) then dir = dir + right end
    if input:GetKeyDown(KEY_A) then dir = dir - right end

    if dir:Length() > 0.01 then
        dir = dir:Normalized()
    end
    return dir
end

function PlayerMovement._startDash(cameraYaw)
    state_ = "dashing"
    dashTimer_ = Config.DASH_DURATION
    dashCooldown_ = DASH_CD

    -- 冲刺方向 = 当前移动方向，无输入时用相机前方
    dashDir_ = PlayerMovement._getMoveDirection(cameraYaw)
    if dashDir_:Length() < 0.01 then
        local yawRad = math.rad(cameraYaw)
        dashDir_ = Vector3(math.sin(yawRad), 0, math.cos(yawRad))
    end

    -- 取消垂直速度
    local vel = PlayerPhysics.GetVelocity()
    vel.y = 0
    PlayerPhysics.SetVelocity(vel)

    -- TODO: 播放冲刺音效 + FOV 拉宽
end

function PlayerMovement._updateDash(dt)
    dashTimer_ = dashTimer_ - dt

    -- 冲刺期间直接设置速度（忽略重力）
    PlayerPhysics.SetVelocity(Vector3(
        dashDir_.x * Config.DASH_SPEED,
        0,
        dashDir_.z * Config.DASH_SPEED
    ))

    if dashTimer_ <= 0 then
        state_ = "normal"
        -- 冲刺结束，保留一部分动量
        local vel = PlayerPhysics.GetVelocity()
        vel.x = vel.x * 0.3
        vel.z = vel.z * 0.3
        PlayerPhysics.SetVelocity(vel)
    end
end

function PlayerMovement._updateClimb(dt, cameraYaw)
    -- 攀爬：上升 + 可以左右移动
    local vel = Vector3(0, Config.CLIMB_SPEED, 0)

    local yawRad = math.rad(cameraYaw)
    local right = Vector3(math.cos(yawRad), 0, -math.sin(yawRad))

    if input:GetKeyDown(KEY_A) then vel = vel - right * 5 end
    if input:GetKeyDown(KEY_D) then vel = vel + right * 5 end

    PlayerPhysics.SetVelocity(vel)

    -- 离开墙壁时退出攀爬
    if not PlayerPhysics.IsOnWall() or PlayerPhysics.IsOnGround() then
        state_ = "normal"
    end
end

function PlayerMovement.GetState()
    return state_
end

return PlayerMovement
```

## 实现：Mechanics.lua（轨道/气球）

```lua
-- scripts/world/Mechanics.lua
local Config = require("scripts/config")
local PlayerPhysics = require("scripts/player/PlayerPhysics")

local Mechanics = {}

-- 轨道数据
local railCurves_ = {}      -- { points = {Vector3, ...}, samples = {Vector3, ...} }
local currentRail_ = nil    -- 当前轨道索引
local railProgress_ = 0     -- 0~1 进度
local RAIL_SPEED = 30.0     -- 轨道移动速度（单位/秒）

-- 气球数据
local balloons_ = {}        -- { center = Vector3, radius = number }

-- === 轨道系统 ===

function Mechanics.RegisterRail(points)
    -- points: Lua table of Vector3 控制点
    -- 预采样 40 个点用于距离检测
    local samples = {}
    local n = #points
    for i = 0, 39 do
        local t = i / 39
        local p = Mechanics._catmullRomPoint(points, t)
        samples[#samples + 1] = p
    end
    railCurves_[#railCurves_ + 1] = { points = points, samples = samples }
end

function Mechanics.CheckRailProximity(playerPos)
    local DETECT_RANGE_FIRST = 4.0
    local DETECT_RANGE = 2.5

    for ci, curve in ipairs(railCurves_) do
        for si, sample in ipairs(curve.samples) do
            local range = (si <= 3) and DETECT_RANGE_FIRST or DETECT_RANGE
            local dx = playerPos.x - sample.x
            local dy = playerPos.y - sample.y
            local dz = playerPos.z - sample.z
            local distSq = dx*dx + dy*dy + dz*dz
            if distSq < range * range then
                return { curveIndex = ci, t = (si - 1) / 39 }
            end
        end
    end
    return nil
end

function Mechanics.EnterRail(curveIndex, t)
    currentRail_ = curveIndex
    railProgress_ = t
end

function Mechanics.UpdateRail(dt)
    if not currentRail_ then return end

    local curve = railCurves_[currentRail_]
    local totalLength = Mechanics._estimateCurveLength(curve.points)
    local speed = RAIL_SPEED / totalLength  -- 归一化速度
    railProgress_ = railProgress_ + speed * dt

    if railProgress_ >= 1.0 then
        Mechanics.ExitRail()
        return
    end

    -- 设置玩家位置到曲线上
    local pos = Mechanics._catmullRomPoint(curve.points, railProgress_)
    local tangent = Mechanics._catmullRomTangent(curve.points, railProgress_)

    -- 偏移：玩家在轨道上方
    pos.y = pos.y + 0.5

    PlayerPhysics.SetVelocity(tangent * RAIL_SPEED)
    -- 直接设位置（轨道期间不做碰撞）
    local node = PlayerPhysics.GetNode()
    node.position = pos
end

function Mechanics.ExitRail()
    currentRail_ = nil
    railProgress_ = 0
end

-- === 气球系统 ===

function Mechanics.RegisterBalloon(center, radius)
    balloons_[#balloons_ + 1] = { center = center, radius = radius or 2.0 }
end

function Mechanics.CheckBalloonCollision(playerPos)
    for i, balloon in ipairs(balloons_) do
        local dx = playerPos.x - balloon.center.x
        local dy = playerPos.y - balloon.center.y
        local dz = playerPos.z - balloon.center.z
        local distSq = dx*dx + dy*dy + dz*dz
        if distSq < balloon.radius * balloon.radius then
            -- 只有从上方落下时触发
            if PlayerPhysics.GetVelocity().y < 0 then
                return true
            end
        end
    end
    return false
end

-- === Catmull-Rom 数学 ===

function Mechanics._catmullRomPoint(points, t)
    local n = #points
    local f = t * (n - 1)
    local i = math.floor(f)
    local frac = f - i

    -- 四个控制点（钳位边界）
    local p0 = points[math.max(1, i)]
    local p1 = points[math.max(1, math.min(n, i + 1))]
    local p2 = points[math.max(1, math.min(n, i + 2))]
    local p3 = points[math.max(1, math.min(n, i + 3))]

    local t2 = frac * frac
    local t3 = t2 * frac

    -- Catmull-Rom 插值
    local x = 0.5 * ((2*p1.x) + (-p0.x + p2.x)*frac +
              (2*p0.x - 5*p1.x + 4*p2.x - p3.x)*t2 +
              (-p0.x + 3*p1.x - 3*p2.x + p3.x)*t3)
    local y = 0.5 * ((2*p1.y) + (-p0.y + p2.y)*frac +
              (2*p0.y - 5*p1.y + 4*p2.y - p3.y)*t2 +
              (-p0.y + 3*p1.y - 3*p2.y + p3.y)*t3)
    local z = 0.5 * ((2*p1.z) + (-p0.z + p2.z)*frac +
              (2*p0.z - 5*p1.z + 4*p2.z - p3.z)*t2 +
              (-p0.z + 3*p1.z - 3*p2.z + p3.z)*t3)

    return Vector3(x, y, z)
end

function Mechanics._catmullRomTangent(points, t)
    local delta = 0.001
    local p1 = Mechanics._catmullRomPoint(points, math.max(0, t - delta))
    local p2 = Mechanics._catmullRomPoint(points, math.min(1, t + delta))
    return (p2 - p1):Normalized()
end

function Mechanics._estimateCurveLength(points)
    local length = 0
    local prev = Mechanics._catmullRomPoint(points, 0)
    for i = 1, 20 do
        local t = i / 20
        local curr = Mechanics._catmullRomPoint(points, t)
        length = length + (curr - prev):Length()
        prev = curr
    end
    return length
end

return Mechanics
```

## 移动状态总结

| 状态 | 触发条件 | 行为 | 退出条件 |
|------|----------|------|----------|
| normal | 默认 | WASD移动+重力+跳跃 | 进入其他状态 |
| dashing | Shift按下 | 固定方向高速，忽略重力 | dashTimer <= 0 |
| wallSlide | 贴墙+空中 | 限制下落速度为-5 | 着地或离墙 |
| climbing | 贴墙+按W | 向上+左右移动 | 离墙/着地/按Space |
| onRail | 进入轨道检测范围 | 沿曲线移动 | progress>=1 或 Space |

## 关键移植注意点

1. **分轴碰撞顺序**: 必须 X → Z → Y，与原游戏一致
2. **二段跳恢复**: 着地时 `jumpsLeft = 2`，气球弹跳也恢复
3. **冲刺不受重力**: 冲刺期间 Y 速度强制为 0
4. **贴墙跳角度**: 从墙面弹开 + 向上，wallNormal * 12 + jumpForce
5. **轨道退出动量**: 退出轨道时保留切线方向速度
6. **Catmull-Rom 曲线**: 使用 4 控制点插值，边界钳位
