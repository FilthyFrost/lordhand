# 08 — UI 系统

## 概述

原游戏 UI 分三层：
1. **主菜单** — Dreamcore 风格 (渐变背景 + Chrome 金属文字 + 赛博符文)
2. **设置面板** — 音量、FOV、血腥效果开关
3. **战斗 HUD** — 生命值、准星、高度显示、弹药（针）指示

MAKER 使用 `urhox-libs/UI` (Yoga Flexbox + NanoVG) 实现 UI。

```lua
            { family = "title", weights = { normal = "Fonts/MiSans-Regular.ttf" } }
        },
        scale = UI.Scale.DEFAULT,
    })

    MainMenu._build()
end

function MainMenu._build()

    local titleLabel = UI.Label {
        id = "title",
        text = "LORDHAND",

        fontColor = {200, 200, 210},
        fontFamily = "title",
    }

    -- 副标题
    local subtitleLabel = UI.Label {
        id = "subtitle",
        text = "SURREAL CATHEDRAL",
        fontSize = 16,
        fontColor = {100, 100, 120},
    }

    -- 按钮
    local newGameBtn = UI.Button {
        id = "btn_new",
        text = "NEW GAME",
        fontSize = 20,
        fontColor = {180, 180, 190},
        width = 200, height = 50,
        onClick = function()
            MainMenu.Hide()
            if onNewGame_ then onNewGame_() end
        end,
    }

    local continueBtn = UI.Button {
        id = "btn_continue",
        text = "CONTINUE",
        fontSize = 20,
        fontColor = {120, 120, 130},
        width = 200, height = 50,
        onClick = function()
            MainMenu.Hide()
            if onContinue_ then onContinue_() end
        end,
    }

    local settingsBtn = UI.Button {
        id = "btn_settings",

        fontSize = 20,

        width = 200, height = 50,
        onClick = function()

        width = "100%", height = "100%",

        alignItems = "center",
        justifyContent = "center",

            titleLabel,
            subtitleLabel,
            UI.Panel { height = 60 },  -- 间距
            newGameBtn,
            UI.Panel { height = 15 },
            continueBtn,
            UI.Panel { height = 15 },
            settingsBtn,
        }
    }

    UI.SetRoot(rootPanel_)
end

function MainMenu._showSettings()
    -- 设置面板（覆盖在菜单上）
    -- FOV 滑块
    -- 音量滑块
    -- 血腥效果开关
    -- 返回按钮
    -- 具体实现取决于 UI.Slider / UI.Toggle 组件可用性
end

function MainMenu.Show()
    visible_ = true
    if rootPanel_ then
        rootPanel_:SetVisible(true)
    end
    SampleInitMouseMode(MM_ABSOLUTE)  -- 菜单中显示鼠标
end

function MainMenu.Hide()
    visible_ = false
    if rootPanel_ then
        rootPanel_:SetVisible(false)
    end
    SampleInitMouseMode(MM_RELATIVE)  -- 游戏中隐藏鼠标
end

function MainMenu.IsVisible()
    return visible_
end

return MainMenu
```

```lua
-- scripts/ui/BattleHUD.lua
local UI = require("urhox-libs/UI")
local Config = require("scripts/config")

local BattleHUD = {}

local rootPanel_ = nil
local healthLabel_ = nil
local healthBar_ = nil
local needleLabels_ = {}  -- 6个针指示器
local heightLabel_ = nil
local crosshair_ = nil

function BattleHUD.Init()
    -- 生命值（左下）
    healthLabel_ = UI.Label {
        id = "hp_text",
        text = "100",
        fontSize = 14,
        fontColor = {255, 60, 60},
    }

    -- 生命条（红色矩形，宽度随 HP 变化）
    healthBar_ = UI.Panel {
        id = "hp_bar",
        width = 200, height = 8,
        backgroundColor = {220, 30, 30, 200},
    }

    local healthBarBg = UI.Panel {
        width = 200, height = 8,
        backgroundColor = {50, 10, 10, 150},
        children = { healthBar_ }
    }

    -- 针指示器（底部中央，6个点）
    local needleContainer = UI.Panel {
        flexDirection = "row",
        alignItems = "center",
        children = {}
    }
    for i = 1, 6 do
        local needle = UI.Panel {
            id = "needle_" .. i,
            width = 8, height = 24,
            backgroundColor = {0, 255, 255, 200},
            marginLeft = 4, marginRight = 4,
        }
        needleLabels_[i] = needle
        -- 添加到容器
        -- (注: 具体 children 动态构建取决于 UI API)
    end

    -- 高度显示（右上）
    heightLabel_ = UI.Label {
        id = "height",
        text = "ALT: 0",
        fontSize = 12,
        fontColor = {150, 150, 160},
    }

    -- 准星（屏幕中心）
    crosshair_ = UI.Panel {
        id = "crosshair",
        width = 2, height = 2,
        backgroundColor = {255, 255, 255, 180},
        position = "absolute",
    }

    -- HUD 根面板
    rootPanel_ = UI.Panel {
        width = "100%", height = "100%",
        pointerEvents = "box-none",  -- 不阻挡点击
        children = {
            -- 左下：生命值
            UI.Panel {
                position = "absolute",
                bottom = 30, left = 20,
                children = { healthLabel_, healthBarBg }
            },
            -- 右上：高度
            UI.Panel {
                position = "absolute",
                top = 20, right = 20,
                children = { heightLabel_ }
            },
            -- 底部中央：针
            UI.Panel {
                position = "absolute",
                bottom = 50,
                left = "50%",  -- 居中
                children = { needleContainer }
            },
            -- 中心：准星
            UI.Panel {
                position = "absolute",
                top = "50%", left = "50%",
                children = { crosshair_ }
            },
        }
    }

    UI.SetRoot(rootPanel_)
end

function BattleHUD.Update(hp, needleCount, altitude)
    -- 更新生命值
    if healthLabel_ then
        healthLabel_:SetText(tostring(math.floor(hp)))
    end
    if healthBar_ then
        local hpRatio = hp / Config.MAX_HP
        -- 动态宽度（200 * ratio）
        -- 注: 需要 UI 支持动态 width 更新
    end

    -- 更新针指示器
    for i = 1, 6 do
        if needleLabels_[i] then
            if i <= needleCount then
                -- 亮色（有针）
                -- needleLabels_[i]:SetBackgroundColor({0, 255, 255, 200})
            else
                -- 暗色（无针）
                -- needleLabels_[i]:SetBackgroundColor({30, 60, 60, 100})
            end
        end
    end

    -- 更新高度
    if heightLabel_ then
        heightLabel_:SetText("ALT: " .. math.floor(altitude))
    end
end

function BattleHUD.Show()
    if rootPanel_ then rootPanel_:SetVisible(true) end
end

function BattleHUD.Hide()
    if rootPanel_ then rootPanel_:SetVisible(false) end
end
```

## UI 视觉风格说明

### 原始风格特征
- **背景**: 径向渐变 (深紫→暗蓝→黑)
- **文字**: Chrome 金属效果 (线性渐变 + 模糊阴影)
- **装饰**: Cybersigilism 符文图案 (随鼠标视差)
- **CRT**: 扫描线叠加

### MAKER 中的近似方案
- **背景**: `backgroundColor` 纯色深暗即可（NanoVG 不支持径向渐变）
- **文字**: 白/灰色文字，加粗表示重要
- **装饰**: 省略（NanoVG 能力有限）
- **CRT**: 省略

```lua
-- ESC 键暂停
function TogglePause()
    if gameState == "playing" then
        gameState = "paused"
        SampleInitMouseMode(MM_ABSOLUTE)
        -- 显示暂停UI
    elseif gameState == "paused" then
        gameState = "playing"
        SampleInitMouseMode(MM_RELATIVE)
        -- 隐藏暂停UI
    end
end
```

## 设置项数据绑定

| 设置项 | 类型 | 范围 | 默认值 | 作用 |
|--------|------|------|--------|------|
| Master Volume | Slider | 0~100 | 80 | 全局音量 |
| FOV | Slider | 60~140 | 90 | 相机视野角 |
| Blood Effects | Toggle | on/off | on | 血液粒子开关 |

```lua
-- 设置应用
function ApplySettings(settings)
    -- FOV
    Config.FOV_DEFAULT = settings.fov
    -- 音量（MAKER 音频系统）
    audio.masterGain = settings.volume / 100
    -- 血腥效果
    Config.BLOOD_ENABLED = settings.blood
end
```
