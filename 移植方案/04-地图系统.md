# 04 — 地图系统

## 概述

原游戏世界由两层数据构成：
1. **渲染层**: 365 个 `.bin` 预烘焙网格区域（顶点+颜色+索引）
2. **碰撞层**: 184 个 JSON chunk 文件（AABB 盒子集合，72万+碰撞体）

MAKER 移植方案：
- 渲染：将 chunk JSON 数据用 `CustomGeometry` 逐面渲染（类似体素渲染，但尺寸不固定）
- 碰撞：纯 Lua 空间哈希 AABB 检测
- 着色：程序化材质模拟 Acid Holy 风格

---

## 原始数据格式

### chunk_X_Z.json 格式
```json
{
  "cx": 0,
  "cz": 10,
  "boxCount": 7299,
  "boxes": [
    {
      "x": 9.0,
      "y": 86.0,
      "z": 649.0,
      "w": 1,
      "h": 1,
      "d": 1,
      "color": 15131357
    }
  ]
}
```

字段说明：
- `x, y, z`: 盒子中心世界坐标
- `w, h, d`: 盒子宽/高/深（单位 = 1米）
- `color`: RGB 整数（如 15131357 = 0xE6E6DD）

### 颜色转换
```
整数 → RGB:
  r = (color >> 16) & 0xFF → 0~255
  g = (color >> 8) & 0xFF → 0~255
  b = color & 0xFF → 0~255

MAKER Color:
  Color(r/255, g/255, b/255, 1.0)
```

## 数据预处理（转换为 Lua）

将每个 `chunk_X_Z.json` 转换为 `chunk_X_Z.lua`：

```lua
-- assets/Data/chunks/chunk_0_10.lua
return {
    cx = 0, cz = 10,
    boxes = {
        -- {x, y, z, w, h, d, r, g, b}
        {9, 86, 649, 1, 1, 1, 0.902, 0.902, 0.867},
        {10, 86, 649, 1, 1, 1, 0.902, 0.902, 0.867},
        -- ... 7299 条
    }
}
```

注意：Z 坐标需要取反（Three.js 右手系 → MAKER 左手系）：`z_maker = -z_threejs`

## 实现：WorldLoader.lua

```lua
-- scripts/world/WorldLoader.lua
local Config = require("scripts/config")
local ChunkRenderer = require("scripts/world/ChunkRenderer")

local WorldLoader = {}

local allColliders_ = {}  -- 所有碰撞体 {minX,maxX,minY,maxY,minZ,maxZ,type}
local chunkNodes_ = {}    -- 已渲染的 chunk 节点
local scene_ = nil

-- 需要加载的 chunk 列表（从 manifest 获取或硬编码）
local CHUNK_LIST = {}  -- 由数据转换工具生成

function WorldLoader.Load(scene, onComplete)
    scene_ = scene

    -- 加载 chunk 列表（预生成的索引文件）
    local manifest = require("assets/Data/chunks/manifest")
    CHUNK_LIST = manifest.chunks  -- { {cx=0,cz=3}, {cx=0,cz=4}, ... }

    -- 分帧加载：每帧加载 2 个 chunk 避免卡顿
    local loadIndex = 1
    local totalChunks = #CHUNK_LIST

    local function loadBatch()
        local batchEnd = math.min(loadIndex + 1, totalChunks)
        for i = loadIndex, batchEnd do
            local entry = CHUNK_LIST[i]
            local chunkData = require("assets/Data/chunks/chunk_" .. entry.cx .. "_" .. entry.cz)
            WorldLoader._processChunk(chunkData)
        end
        loadIndex = batchEnd + 1

        if loadIndex > totalChunks then
            -- 加载完成
            if onComplete then onComplete() end
        end
        -- 注意：实际分帧需要用协程或计时器，这里简化为同步加载
    end

    -- 同步加载全部（首次可接受，后续可改为分帧）
    for i = 1, totalChunks do
        local entry = CHUNK_LIST[i]
        local ok, chunkData = pcall(require, "assets/Data/chunks/chunk_" .. entry.cx .. "_" .. entry.cz)
        if ok and chunkData then
            WorldLoader._processChunk(chunkData)
        end
    end

    if onComplete then onComplete() end
end

function WorldLoader._processChunk(chunkData)
    local boxes = chunkData.boxes
    if not boxes or #boxes == 0 then return end

    -- 1. 注册碰撞体
    for i = 1, #boxes do
        local b = boxes[i]
        local x, y, z, w, h, d = b[1], b[2], b[3], b[4], b[5], b[6]
        local hw, hh, hd = w * 0.5, h * 0.5, d * 0.5

        allColliders_[#allColliders_ + 1] = {
            minX = x - hw, maxX = x + hw,
            minY = y - hh, maxY = y + hh,
            minZ = z - hd, maxZ = z + hd,
            type = "solid"
        }
    end

    -- 2. 渲染 chunk
    local node = ChunkRenderer.RenderChunk(scene_, chunkData)
    if node then
        chunkNodes_[#chunkNodes_ + 1] = node
    end
end

function WorldLoader.GetColliders()
    return allColliders_
end

function WorldLoader.GetChunkNodes()
    return chunkNodes_
end

return WorldLoader
```

```lua
-- scripts/world/ChunkRenderer.lua
local Config = require("scripts/config")

local ChunkRenderer = {}

-- 材质缓存（按颜色索引）
local materialCache_ = {}

-- 立方体面数据（6面，每面4顶点→2三角形=6顶点）
local CUBE_FACES = {
    -- { normal, vertices: {pos_offset, uv} }
    { -- +Y (top)
        normal = Vector3(0, 1, 0),
        verts = {
            {Vector3(-0.5, 0.5, -0.5)}, {Vector3(-0.5, 0.5, 0.5)},
            {Vector3(0.5, 0.5, 0.5)},   {Vector3(0.5, 0.5, 0.5)},
            {Vector3(0.5, 0.5, -0.5)},  {Vector3(-0.5, 0.5, -0.5)},
        }
    },
    { -- -Y (bottom)
        normal = Vector3(0, -1, 0),
        verts = {
            {Vector3(-0.5, -0.5, 0.5)},  {Vector3(-0.5, -0.5, -0.5)},
            {Vector3(0.5, -0.5, -0.5)},  {Vector3(0.5, -0.5, -0.5)},
            {Vector3(0.5, -0.5, 0.5)},   {Vector3(-0.5, -0.5, 0.5)},
        }
    },
    { -- +X (right)
        normal = Vector3(1, 0, 0),
        verts = {
            {Vector3(0.5, -0.5, -0.5)}, {Vector3(0.5, 0.5, -0.5)},
            {Vector3(0.5, 0.5, 0.5)},   {Vector3(0.5, 0.5, 0.5)},
            {Vector3(0.5, -0.5, 0.5)},  {Vector3(0.5, -0.5, -0.5)},
        }
    },
    { -- -X (left)
        normal = Vector3(-1, 0, 0),
        verts = {
            {Vector3(-0.5, -0.5, 0.5)},  {Vector3(-0.5, 0.5, 0.5)},
            {Vector3(-0.5, 0.5, -0.5)},  {Vector3(-0.5, 0.5, -0.5)},
            {Vector3(-0.5, -0.5, -0.5)}, {Vector3(-0.5, -0.5, 0.5)},
        }
    },
    { -- +Z (front)
        normal = Vector3(0, 0, 1),
        verts = {
            {Vector3(-0.5, -0.5, 0.5)}, {Vector3(0.5, -0.5, 0.5)},
            {Vector3(0.5, 0.5, 0.5)},   {Vector3(0.5, 0.5, 0.5)},
            {Vector3(-0.5, 0.5, 0.5)},  {Vector3(-0.5, -0.5, 0.5)},
        }
    },
    { -- -Z (back)
        normal = Vector3(0, 0, -1),
        verts = {
            {Vector3(0.5, -0.5, -0.5)},  {Vector3(-0.5, -0.5, -0.5)},
            {Vector3(-0.5, 0.5, -0.5)},  {Vector3(-0.5, 0.5, -0.5)},
            {Vector3(0.5, 0.5, -0.5)},   {Vector3(0.5, -0.5, -0.5)},
        }
    },
}

function ChunkRenderer.RenderChunk(scene, chunkData)
    local boxes = chunkData.boxes
    if not boxes or #boxes == 0 then return nil end

    local chunkNode = scene:CreateChild("Chunk_" .. chunkData.cx .. "_" .. chunkData.cz)
    local geometry = chunkNode:CreateComponent("CustomGeometry")
    geometry:BeginGeometry(0, TRIANGLE_LIST)

    local vertexCount = 0

    for i = 1, #boxes do
        local b = boxes[i]
        local cx, cy, cz = b[1], b[2], b[3]
        local w, h, d = b[4], b[5], b[6]
        local r, g, bl = b[7], b[8], b[9]
        local color = Color(r, g, bl, 1.0)

        -- 渲染所有 6 个面（无邻居剔除优化，因为盒子尺寸不一）
        -- 优化：可后续添加邻居检测跳过被遮挡的面
        for _, face in ipairs(CUBE_FACES) do
            for _, vert in ipairs(face.verts) do
                local vpos = vert[1]
                geometry:DefineVertex(Vector3(
                    cx + vpos.x * w,
                    cy + vpos.y * h,
                    cz + vpos.z * d
                ))
                geometry:DefineNormal(face.normal)
                geometry:DefineTexCoord(Vector2(0, 0))  -- 无纹理
                geometry:DefineColor(color)
                vertexCount = vertexCount + 1
            end
        end
    end

    geometry:Commit()

    -- 设置材质
    if vertexCount > 0 then
        geometry:SetMaterial(0, ChunkRenderer._getChunkMaterial())
    end

    return chunkNode
end

function ChunkRenderer._getChunkMaterial()
    if materialCache_["chunk"] then return materialCache_["chunk"] end

    -- 使用带顶点颜色的 PBR 材质
    -- 高粗糙度 + 低金属度 → 接近卡通/哑光效果
    local mat = Material:new()
    mat:SetTechnique(0, cache:GetResource("Technique", "Techniques/DiffVCol.xml"))
    mat:SetShaderParameter("MatDiffColor", Variant(Color(1, 1, 1, 1)))

    materialCache_["chunk"] = mat
    return mat
end

return ChunkRenderer
```

## 实现：SpatialHash.lua

```lua
-- scripts/world/SpatialHash.lua
local Config = require("scripts/config")

local SpatialHash = {}

local CELL_SIZE = Config.SPATIAL_CELL_SIZE  -- 20
local grid_ = {}  -- key: "cx_cy_cz" → {collider, ...}
local floor = math.floor

--- 将世界坐标转换为格子坐标
local function toCell(x, y, z)
    return floor(x / CELL_SIZE), floor(y / CELL_SIZE), floor(z / CELL_SIZE)
end

--- 格子键
local function cellKey(cx, cy, cz)
    return cx .. "_" .. cy .. "_" .. cz
end

--- 构建空间哈希（加载地图后调用一次）
function SpatialHash.Build(colliders)
    grid_ = {}
    for i = 1, #colliders do
        local col = colliders[i]
        -- 计算碰撞体覆盖的所有格子
        local minCX, minCY, minCZ = toCell(col.minX, col.minY, col.minZ)
        local maxCX, maxCY, maxCZ = toCell(col.maxX, col.maxY, col.maxZ)

        for cx = minCX, maxCX do
            for cy = minCY, maxCY do
                for cz = minCZ, maxCZ do
                    local key = cellKey(cx, cy, cz)
                    if not grid_[key] then
                        grid_[key] = {}
                    end
                    local cell = grid_[key]
                    cell[#cell + 1] = col
                end
            end
        end
    end
end

--- AABB 碰撞检查：给定玩家包围盒，返回是否与任何碰撞体重叠
function SpatialHash.CheckCollision(minX, minY, minZ, maxX, maxY, maxZ)
    local minCX, minCY, minCZ = toCell(minX, minY, minZ)
    local maxCX, maxCY, maxCZ = toCell(maxX, maxY, maxZ)

    -- 去重（同一碰撞体可能在多个格子中）
    local checked = {}

    for cx = minCX, maxCX do
        for cy = minCY, maxCY do
            for cz = minCZ, maxCZ do
                local key = cellKey(cx, cy, cz)
                local cell = grid_[key]
                if cell then
                    for i = 1, #cell do
                        local col = cell[i]
                        if not checked[col] then
                            checked[col] = true
                            -- AABB 重叠测试
                            if minX < col.maxX and maxX > col.minX and
                               minY < col.maxY and maxY > col.minY and
                               minZ < col.maxZ and maxZ > col.minZ then
                                return true
                            end
                        end
                    end
                end
            end
        end
    end
    return false
end

--- 获取位置附近的所有碰撞体（用于精确检测）
function SpatialHash.GetNearby(x, y, z)
    local cx, cy, cz = toCell(x, y, z)
    local results = {}
    local checked = {}

    -- 查询 3x3x3 邻域
    for dx = -1, 1 do
        for dy = -1, 1 do
            for dz = -1, 1 do
                local key = cellKey(cx + dx, cy + dy, cz + dz)
                local cell = grid_[key]
                if cell then
                    for i = 1, #cell do
                        local col = cell[i]
                        if not checked[col] then
                            checked[col] = true
                            results[#results + 1] = col
                        end
                    end
                end
            end
        end
    end
    return results
end

--- 获取区域内所有碰撞体（弹射物用）
function SpatialHash.QueryRange(minX, minY, minZ, maxX, maxY, maxZ)
    local minCX, minCY, minCZ = toCell(minX, minY, minZ)
    local maxCX, maxCY, maxCZ = toCell(maxX, maxY, maxZ)
    local results = {}
    local checked = {}

    for cx = minCX, maxCX do
        for cy = minCY, maxCY do
            for cz = minCZ, maxCZ do
                local key = cellKey(cx, cy, cz)
                local cell = grid_[key]
                if cell then
                    for i = 1, #cell do
                        local col = cell[i]
                        if not checked[col] then
                            checked[col] = true
                            if minX < col.maxX and maxX > col.minX and
                               minY < col.maxY and maxY > col.minY and
                               minZ < col.maxZ and maxZ > col.minZ then
                                results[#results + 1] = col
                            end
                        end
                    end
                end
            end
        end
    end
    return results
end

return SpatialHash
```

## Acid Holy Shader 风格模拟

原始着色器效果：
1. **顶点扭曲** (Acid Wobble): `vertex.y += sin(time * 2 + vertex.x * 0.5) * 0.15`
2. **卡通着色** (Cel): 光照用阶梯函数而非平滑渐变
3. **描边** (Outline): 反转法线膨胀的黑色第二遍渲染

### MAKER 中的实现策略

由于 MAKER 不支持自定义 Vertex/Fragment Shader，采用以下近似方案：

**方案 1：每帧微偏移节点（模拟顶点扭曲）**
```lua
-- 全局时间驱动的节点抖动（应用于整个 chunk）
local wobbleTime_ = 0

function ApplyWobble(chunkNode, dt)
    wobbleTime_ = wobbleTime_ + dt * 2
    local offset = math.sin(wobbleTime_) * 0.05
    -- 整体微偏移（粗略近似）
    chunkNode.position = Vector3(0, offset, 0)
end
```

**方案 2：使用 NoTextureUnlit 避免 PBR 真实感**
```lua
-- 无光照材质 → 更接近卡通/flat 效果
local mat = Material:new()
mat:SetTechnique(0, cache:GetResource("Technique", "Techniques/NoTextureUnlit.xml"))
mat:SetShaderParameter("MatDiffColor", Variant(color))
```

**方案 3：描边通过第二层缩放节点**
```lua
-- 为重要物体（敌人、手）添加描边
function CreateOutline(parentNode, scale)
    local outlineNode = parentNode:CreateChild("Outline")
    outlineNode.scale = Vector3(scale, scale, scale)  -- 略大于原模型
    local model = outlineNode:CreateComponent("StaticModel")
    -- 复制相同模型但用纯黑材质
    model:SetModel(parentNode:GetComponent("StaticModel"):GetModel())
    local mat = Material:new()
    mat:SetTechnique(0, cache:GetResource("Technique", "Techniques/NoTextureUnlit.xml"))
    mat:SetShaderParameter("MatDiffColor", Variant(Color(0, 0, 0, 1)))
    model:SetMaterial(mat)
end
```

### 2. 视距裁剪
```lua
-- 只渲染玩家附近的 chunk
local VIEW_DISTANCE = 150  -- 米
function ShouldRenderChunk(chunkCX, chunkCZ, playerPos)
    local chunkCenterX = chunkCX * 64 + 32  -- 假设 chunk 跨度 64
    local chunkCenterZ = chunkCZ * 64 + 32
    local dx = chunkCenterX - playerPos.x
    local dz = chunkCenterZ - playerPos.z
    return (dx*dx + dz*dz) < VIEW_DISTANCE * VIEW_DISTANCE
end
```

### 3. 分帧加载
每帧只构建 1-2 个新 chunk 的 CustomGeometry，避免长时间卡顿。
