# 05 — 战斗系统

## 概述

原游戏战斗系统核心：
- **6 针资源管理**: 6 根针（needle），攻击消耗，每秒恢复 1 根
- **4 种射击模式**: LMB 点射、LMB 长按连射、RMB 点射单针、RMB 长按蓄力大招
- **弹射物系统**: AABB 碰撞检测、多步采样防穿透
- **程序化手掌模型**: FPS 视图中显示的骨骼手掌

---

## 6 针资源系统

```
最大针数: 6
恢复速率: 1针/秒（正常时）
大招后冷却: 6秒（期间不恢复）
```

### 攻击消耗表

| 模式 | 触发 | 消耗 | 冷却 | 说明 |
|------|------|------|------|------|
| 散射 (Shotgun) | LMB 点击 | 0 | 0.5s | 50发散射弹 |
| 连射 (Stream) | LMB 长按 | 0 | 55ms/发 | 持续射击 |
| 单针 (Needle) | RMB 点击 | 1针 | 无 | 巨型穿透弹 |
| 大招 (Ultimate) | RMB 长按→松开 | 6针 | 6s | 瞬间贯穿光束 |

## 实现：WeaponSystem.lua

```lua
-- scripts/combat/WeaponSystem.lua
local Config = require("scripts/config")

local WeaponSystem = {}

-- 状态
local needles_ = 6           -- 当前针数
local regenTimer_ = 0        -- 恢复计时
local ultimateCooldown_ = 0  -- 大招冷却
local shotgunCooldown_ = 0   -- 散射冷却
local streamTimer_ = 0       -- 连射计时

-- 蓄力状态
local isChargingUlt_ = false
local chargeTime_ = 0
local CHARGE_THRESHOLD = 0.5   -- 长按超过此时间视为蓄力
local CHARGE_FULL_TIME = 2.0   -- 蓄满时间

-- 持续射击状态
local isStreaming_ = false
local streamStartTime_ = 0

-- 输入状态
local lmbDown_ = false
local rmbDown_ = false
local prevLMB_ = false
local prevRMB_ = false
local lmbHoldTime_ = 0
local rmbHoldTime_ = 0

-- 引用
local scene_ = nil
local camera_ = nil      -- FPSCamera 模块
local handModel_ = nil   -- HandModel 模块
local projectiles_ = nil -- ProjectileSystem 模块

function WeaponSystem.Init(scene, camera, handModel, projectileSystem)
    scene_ = scene
    camera_ = camera
    handModel_ = handModel
    projectiles_ = projectileSystem
end

function WeaponSystem.Update(dt)
    -- 冷却递减
    if ultimateCooldown_ > 0 then
        ultimateCooldown_ = ultimateCooldown_ - dt
    end
    if shotgunCooldown_ > 0 then
        shotgunCooldown_ = shotgunCooldown_ - dt
    end

    -- 针恢复
    if ultimateCooldown_ <= 0 and needles_ < Config.MAX_NEEDLES then
        regenTimer_ = regenTimer_ + dt
        if regenTimer_ >= Config.NEEDLE_REGEN_INTERVAL then
            regenTimer_ = regenTimer_ - Config.NEEDLE_REGEN_INTERVAL
            needles_ = math.min(needles_ + 1, Config.MAX_NEEDLES)
        end
    end

    -- 读取输入
    local lmbNow = input:GetMouseButtonDown(MOUSEB_LEFT)
    local rmbNow = input:GetMouseButtonDown(MOUSEB_RIGHT)
    local lmbPressed = lmbNow and not prevLMB_
    local lmbReleased = not lmbNow and prevLMB_
    local rmbPressed = rmbNow and not prevRMB_
    local rmbReleased = not rmbNow and prevRMB_

    -- 长按计时
    if lmbNow then lmbHoldTime_ = lmbHoldTime_ + dt
    else lmbHoldTime_ = 0 end

    if rmbNow then rmbHoldTime_ = rmbHoldTime_ + dt
    else rmbHoldTime_ = 0 end

    prevLMB_ = lmbNow
    prevRMB_ = rmbNow

    -- === LMB 逻辑 ===
    if lmbPressed and shotgunCooldown_ <= 0 then
        WeaponSystem._fireShotgun()
        shotgunCooldown_ = Config.SHOTGUN_COOLDOWN
        isStreaming_ = false
    end

    if lmbNow and lmbHoldTime_ > 0.2 then
        -- 进入连射模式
        isStreaming_ = true
        streamTimer_ = streamTimer_ + dt
        if streamTimer_ >= Config.STREAM_INTERVAL then
            streamTimer_ = streamTimer_ - Config.STREAM_INTERVAL
            WeaponSystem._fireStream()
        end
    end

    if lmbReleased then
        isStreaming_ = false
        streamTimer_ = 0
    end

    -- === RMB 逻辑 ===
    if rmbPressed then
        if needles_ >= 1 then
            isChargingUlt_ = false
            chargeTime_ = 0
        end
    end

    if rmbNow and rmbHoldTime_ > CHARGE_THRESHOLD and needles_ >= 6 then
        -- 蓄力大招
        isChargingUlt_ = true
        chargeTime_ = chargeTime_ + dt
    end

    if rmbReleased then
        if isChargingUlt_ and needles_ >= 6 then
            -- 释放大招
            WeaponSystem._fireUltimate()
            needles_ = 0
            ultimateCooldown_ = Config.ULTIMATE_COOLDOWN
            regenTimer_ = 0
        elseif rmbHoldTime_ < CHARGE_THRESHOLD and needles_ >= 1 then
            -- 短按 = 单针
            WeaponSystem._fireNeedle()
            needles_ = needles_ - 1
        end
        isChargingUlt_ = false
        chargeTime_ = 0
    end
end

-- === 射击方法 ===

function WeaponSystem._fireShotgun()
    local origin = camera_.GetForward and handModel_.GetFireOrigin() or Vector3.ZERO
    local dir = camera_.GetForward()
    local right = camera_.GetRight()
    local up = Vector3(0, 1, 0)

    -- 50 发散射弹，锥形扩散
    local SPREAD = 0.15  -- 弧度
    for i = 1, 50 do
        local spreadX = (math.random() - 0.5) * 2 * SPREAD
        local spreadY = (math.random() - 0.5) * 2 * SPREAD
        local shotDir = (dir + right * spreadX + up * spreadY):Normalized()
        projectiles_.Spawn({
            origin = origin,
            direction = shotDir,
            speed = 80 + math.random() * 20,
            lifetime = 1.5,
            damage = 2,
            type = "dagger",
            color = Config.COLOR_CYAN,
        })
    end

    -- 触发反馈
    camera_.Shake(0.3)
    handModel_.Recoil(0.8)
    -- TODO: 播放散射音效
end

function WeaponSystem._fireStream()
    local origin = handModel_.GetFireOrigin()
    local dir = camera_.GetForward()
    local right = camera_.GetRight()
    local up = Vector3(0, 1, 0)

    -- 4 发 + 轻微扩散
    for i = 1, 4 do
        local spreadX = (math.random() - 0.5) * 0.06
        local spreadY = (math.random() - 0.5) * 0.06
        local shotDir = (dir + right * spreadX + up * spreadY):Normalized()
        projectiles_.Spawn({
            origin = origin,
            direction = shotDir,
            speed = 60,
            lifetime = 2.0,
            damage = 1,
            type = "laser_bolt",
            color = Config.COLOR_WHITE,
        })
    end

    camera_.Shake(0.05)
    handModel_.Recoil(0.2)
end

function WeaponSystem._fireNeedle()
    local origin = handModel_.GetFireOrigin()
    local dir = camera_.GetForward()

    projectiles_.Spawn({
        origin = origin,
        direction = dir,
        speed = 150,
        lifetime = 2.5,
        damage = 15,
        type = "needle",
        color = Config.COLOR_WHITE,
        scale = 3.0,
        piercing = true,
    })

    camera_.Shake(0.5)
    camera_.Impulse(dir * -1, 0.3)
    handModel_.Recoil(1.5)
    -- TODO: 播放单针音效
end

function WeaponSystem._fireUltimate()
    local origin = handModel_.GetFireOrigin()
    local dir = camera_.GetForward()

    -- 大招 = 即时射线（hitscan），不是弹射物
    -- 射线检测所有命中的敌人和环境
    local RAY_LENGTH = 200
    local endPoint = origin + dir * RAY_LENGTH

    -- 检测射线路径上的敌人
    local DemonRegistry = require("scripts/enemies/DemonRegistry")
    local hitDemons = DemonRegistry.RaycastAll(origin, dir, RAY_LENGTH)
    for _, demon in ipairs(hitDemons) do
        demon:TakeDamage(50, dir)
    end

    -- 视觉：生成圆柱体光束（持续 0.5 秒后淡出）
    WeaponSystem._spawnBeamVisual(origin, endPoint)

    -- 反馈
    camera_.Shake(2.0)
    camera_.Impulse(dir * -1, 1.0)
    camera_.SetFOVOffset(15)  -- FOV 拉宽
    handModel_.Recoil(3.0)
    -- TODO: 播放大招音效 + 屏幕闪白
end

function WeaponSystem._spawnBeamVisual(start, finish)
    -- 用圆柱体表示光束
    local beamNode = scene_:CreateChild("UltBeam")
    local mid = (start + finish) * 0.5
    beamNode.position = mid

    local dir = (finish - start):Normalized()
    local length = (finish - start):Length()

    -- 朝向
    beamNode.rotation = Quaternion(Vector3(0, 1, 0), dir)

    local model = beamNode:CreateComponent("StaticModel")
    model:SetModel(cache:GetResource("Model", "Models/Cylinder.mdl"))
    beamNode.scale = Vector3(0.3, length, 0.3)

    -- 发光白色材质
    local mat = Material:new()
    mat:SetTechnique(0, cache:GetResource("Technique", "Techniques/NoTextureUnlit.xml"))
    mat:SetShaderParameter("MatDiffColor", Variant(Color(1, 1, 1, 1)))
    model:SetMaterial(mat)

    -- 0.5 秒后删除
    -- 使用延迟删除（通过 Update 计时或 ScriptObject）
    beamNode:SetVar(StringHash("lifetime"), Variant(0.5))
end

-- === 公共接口 ===

function WeaponSystem.GetNeedleCount()
    return needles_
end

function WeaponSystem.GetState()
    if isChargingUlt_ then return "charge" end
    if isStreaming_ then return "stream" end
    if shotgunCooldown_ > 0.3 then return "fire" end
    return "idle"
end

function WeaponSystem.IsCharging()
    return isChargingUlt_
end

function WeaponSystem.GetChargeProgress()
    if not isChargingUlt_ then return 0 end
    return math.min(1.0, chargeTime_ / CHARGE_FULL_TIME)
end

return WeaponSystem
```

```lua
-- scripts/combat/ProjectileSystem.lua
local Config = require("scripts/config")

local ProjectileSystem = {}

local projectiles_ = {}   -- 活跃弹射物列表
local scene_ = nil
local spatialHash_ = nil
local demonRegistry_ = nil

-- 几何体缓存
local geometryCache_ = {}  -- type → Model resource path

function ProjectileSystem.Init(scene, spatialHash, demonRegistry)
    scene_ = scene
    spatialHash_ = spatialHash
    demonRegistry_ = demonRegistry
end

--- 生成弹射物
---@param params table {origin, direction, speed, lifetime, damage, type, color, scale, piercing, homing}
function ProjectileSystem.Spawn(params)
    if #projectiles_ >= Config.MAX_PROJECTILES then
        -- 移除最老的
        ProjectileSystem._removeProjectile(1)
    end

    local node = scene_:CreateChild("Projectile")
    node.position = params.origin

    -- 视觉模型
    local model = node:CreateComponent("StaticModel")
    local modelPath = ProjectileSystem._getModelPath(params.type)
    model:SetModel(cache:GetResource("Model", modelPath))

    -- 材质
    local mat = Material:new()
    mat:SetTechnique(0, cache:GetResource("Technique", "Techniques/NoTextureUnlit.xml"))
    local c = params.color or Config.COLOR_CYAN
    mat:SetShaderParameter("MatDiffColor", Variant(c))
    model:SetMaterial(mat)

    -- 缩放
    local s = params.scale or 0.15
    node.scale = Vector3(s, s, s)

    -- 朝向
    node.rotation = Quaternion(Vector3(0, 0, 1), params.direction)

    -- 注册
    projectiles_[#projectiles_ + 1] = {
        node = node,
        pos = params.origin:__clone(),
        dir = params.direction:Normalized(),
        speed = params.speed,
        lifetime = params.lifetime,
        age = 0,
        damage = params.damage or 1,
        type = params.type,
        piercing = params.piercing or false,
        homing = params.homing,  -- 可选追踪目标
        radius = s * 0.5,        -- 碰撞半径
    }
end

--- 每帧更新所有弹射物
function ProjectileSystem.Update(dt)
    local hitResults = {}
    local toRemove = {}

    for i = #projectiles_, 1, -1 do
        local p = projectiles_[i]
        p.age = p.age + dt

        -- 超时移除
        if p.age >= p.lifetime then
            toRemove[#toRemove + 1] = i
        else
            -- 追踪（如有）
            if p.homing then
                local toTarget = (p.homing - p.pos):Normalized()
                p.dir = (p.dir + toTarget * 3.0 * dt):Normalized()
            end

            -- 移动（多步采样防穿透）
            local moveDistance = p.speed * dt
            local steps = math.max(1, math.ceil(moveDistance / 2.0))  -- 每 2 单位一步
            local stepDist = moveDistance / steps
            local hit = false

            for step = 1, steps do
                local newPos = p.pos + p.dir * stepDist

                -- 环境碰撞
                local collides = spatialHash_.CheckCollision(
                    newPos.x - p.radius, newPos.y - p.radius, newPos.z - p.radius,
                    newPos.x + p.radius, newPos.y + p.radius, newPos.z + p.radius
                )

                if collides then
                    hitResults[#hitResults + 1] = {
                        type = "environment",
                        position = newPos:__clone(),
                        normal = ProjectileSystem._calcHitNormal(p.pos, newPos),
                        damage = p.damage,
                    }
                    if not p.piercing then
                        hit = true
                        break
                    end
                end

                -- 敌人碰撞
                local hitDemon = demonRegistry_.CheckHit(newPos, p.radius)
                if hitDemon then
                    hitResults[#hitResults + 1] = {
                        type = "enemy",
                        position = newPos:__clone(),
                        normal = (newPos - hitDemon:GetPosition()):Normalized(),
                        damage = p.damage,
                        demon = hitDemon,
                    }
                    hitDemon:TakeDamage(p.damage, p.dir)
                    if not p.piercing then
                        hit = true
                        break
                    end
                end

                p.pos = newPos
            end

            if hit then
                toRemove[#toRemove + 1] = i
            else
                -- 更新节点位置
                p.node.position = p.pos
            end
        end
    end

    -- 移除（倒序保安全）
    table.sort(toRemove, function(a, b) return a > b end)
    for _, idx in ipairs(toRemove) do
        ProjectileSystem._removeProjectile(idx)
    end

    return hitResults
end

function ProjectileSystem._removeProjectile(index)
    local p = projectiles_[index]
    if p and p.node then
        p.node:Remove()
    end
    table.remove(projectiles_, index)
end

function ProjectileSystem._getModelPath(ptype)
    if ptype == "dagger" then return "Models/Cone.mdl" end
    if ptype == "laser_bolt" then return "Models/Cylinder.mdl" end
    if ptype == "needle" then return "Models/Cylinder.mdl" end
    if ptype == "pulse" then return "Models/Sphere.mdl" end
    return "Models/Sphere.mdl"
end

function ProjectileSystem._calcHitNormal(oldPos, newPos)
    -- 简化：基于移动方向取反
    local dir = (newPos - oldPos):Normalized()
    -- 判断哪个轴移动最大
    local ax = math.abs(dir.x)
    local ay = math.abs(dir.y)
    local az = math.abs(dir.z)

    if ax > ay and ax > az then
        return Vector3(dir.x > 0 and -1 or 1, 0, 0)
    elseif ay > az then
        return Vector3(0, dir.y > 0 and -1 or 1, 0)
    else
        return Vector3(0, 0, dir.z > 0 and -1 or 1)
    end
end

function ProjectileSystem.GetCount()
    return #projectiles_
end

return ProjectileSystem
```

```lua
-- scripts/combat/HandModel.lua (简化版)
local Config = require("scripts/config")

local HandModel = {}

local handNode_ = nil       -- 手掌根节点
local fingers_ = {}          -- 5个手指节点组
local coreNode_ = nil        -- 掌心核心
local fireOrigin_ = nil      -- 射击原点

-- 动画状态
local currentState_ = "idle"
local recoilZ_ = 0
local recoilDecay_ = 15.0
local trembleAmount_ = 0

-- 手指动画目标
local STATES = {
    idle    = { curl = 0.3, spread = 0.0, glow = 0.3 },
    fire    = { curl = 0.0, spread = 0.4, glow = 1.0 },
    stream  = { curl = 0.2, spread = 0.3, glow = 0.8 },
    charge  = { curl = 0.8, spread = 0.0, glow = 1.5 },
    release = { curl = 0.0, spread = 0.0, glow = 2.0 },
}

function HandModel.Init(scene, cameraNode)
    -- 手掌挂载在相机子节点（FPS视图）
    handNode_ = cameraNode:CreateChild("Hand")
    handNode_.position = Vector3(0.15, -0.2, -0.4)  -- 右下偏移

    -- 掌心
    local palmNode = handNode_:CreateChild("Palm")
    local palmModel = palmNode:CreateComponent("StaticModel")
    palmModel:SetModel(cache:GetResource("Model", "Models/Cylinder.mdl"))
    palmNode.scale = Vector3(0.08, 0.12, 0.06)
    local palmMat = Material:new()
    palmMat:SetTechnique(0, cache:GetResource("Technique", "Techniques/PBR/PBRNoTexture.xml"))
    palmMat:SetShaderParameter("MatDiffColor", Variant(Color(0.8, 0.8, 0.85, 1)))
    palmMat:SetShaderParameter("Roughness", Variant(0.3))
    palmMat:SetShaderParameter("Metallic", Variant(0.7))
    palmModel:SetMaterial(palmMat)

    -- 核心（八面体）
    coreNode_ = handNode_:CreateChild("Core")
    local coreModel = coreNode_:CreateComponent("StaticModel")
    coreModel:SetModel(cache:GetResource("Model", "Models/Sphere.mdl"))
    coreNode_.scale = Vector3(0.03, 0.03, 0.03)
    coreNode_.position = Vector3(0, 0, -0.05)
    local coreMat = Material:new()
    coreMat:SetTechnique(0, cache:GetResource("Technique", "Techniques/NoTextureUnlit.xml"))
    coreMat:SetShaderParameter("MatDiffColor", Variant(Config.COLOR_CYAN))
    coreModel:SetMaterial(coreMat)

    -- 5 个手指（简化为圆柱段）
    local fingerOffsets = {
        {x = -0.04, z = -0.08, name = "Thumb"},
        {x = -0.02, z = -0.12, name = "Index"},
        {x = 0.0,   z = -0.13, name = "Middle"},
        {x = 0.02,  z = -0.12, name = "Ring"},
        {x = 0.04,  z = -0.10, name = "Pinky"},
    }

    for fi, fdata in ipairs(fingerOffsets) do
        local fingerGroup = handNode_:CreateChild(fdata.name)
        fingerGroup.position = Vector3(fdata.x, 0.05, fdata.z)
        local segments = {}
        for seg = 1, 3 do
            local segNode = fingerGroup:CreateChild("Seg" .. seg)
            segNode.position = Vector3(0, 0, -0.03 * seg)
            segNode.scale = Vector3(0.012, 0.012, 0.03)
            local segModel = segNode:CreateComponent("StaticModel")
            segModel:SetModel(cache:GetResource("Model", "Models/Cylinder.mdl"))
            segModel:SetMaterial(palmMat)
            segments[seg] = segNode
        end
        fingers_[fi] = { node = fingerGroup, segments = segments }
    end

    -- 射击原点 = 中指尖端（世界坐标在 Update 中计算）
    fireOrigin_ = Vector3.ZERO
end

function HandModel.Update(dt, weaponState)
    currentState_ = weaponState or "idle"
    local target = STATES[currentState_] or STATES.idle

    -- 手指卷曲动画（X轴旋转）
    for fi, finger in ipairs(fingers_) do
        for seg, segNode in ipairs(finger.segments) do
            local targetAngle = target.curl * 30 * seg  -- 每节递增
            local currentRot = segNode.rotation:EulerAngles()
            local newX = Lerp(currentRot.x, targetAngle, 8 * dt)
            segNode.rotation = Quaternion(newX, 0, 0)
        end
        -- 手指展开（Y轴旋转）
        local spreadAngle = target.spread * 15 * (fi - 3)  -- 中心为0，两侧展开
        local currentRot = finger.node.rotation:EulerAngles()
        local newY = Lerp(currentRot.y, spreadAngle, 8 * dt)
        finger.node.rotation = Quaternion(0, newY, 0)
    end

    -- 核心脉动
    local pulse = 1.0 + math.sin(os.clock() * 5) * 0.1 * target.glow
    coreNode_.scale = Vector3(0.03 * pulse, 0.03 * pulse, 0.03 * pulse)

    -- 后坐力衰减
    recoilZ_ = recoilZ_ * math.exp(-recoilDecay_ * dt)
    handNode_.position = Vector3(0.15, -0.2, -0.4 + recoilZ_)

    -- 蓄力时手部颤抖
    if currentState_ == "charge" then
        trembleAmount_ = math.min(trembleAmount_ + dt * 2, 1.0)
        local tx = (math.random() - 0.5) * 0.005 * trembleAmount_
        local ty = (math.random() - 0.5) * 0.005 * trembleAmount_
        handNode_.position = handNode_.position + Vector3(tx, ty, 0)
    else
        trembleAmount_ = 0
    end

    -- 更新射击原点（中指尖端世界位置）
    local middleFinger = fingers_[3]
    if middleFinger and middleFinger.segments[3] then
        fireOrigin_ = middleFinger.segments[3]:GetWorldPosition()
    end
end

function HandModel.Recoil(intensity)
    recoilZ_ = recoilZ_ + intensity * 0.05
end

function HandModel.GetFireOrigin()
    return fireOrigin_
end

return HandModel
```

## 大招（Ultimate）射线检测

大招不使用弹射物系统，而是瞬间射线：

```lua
-- 射线 vs AABB 检测（遍历路径上所有敌人）
function RaycastAll(origin, dir, maxDist)
    local hits = {}
    local demons = DemonRegistry.GetAll()
    for _, demon in ipairs(demons) do
        local dPos = demon:GetPosition()
        -- 点到射线的距离
        local toD = dPos - origin
        local proj = toD:DotProduct(dir)
        if proj > 0 and proj < maxDist then
            local closest = origin + dir * proj
            local dist = (closest - dPos):Length()
            if dist < demon:GetRadius() then
                hits[#hits + 1] = demon
            end
        end
    end
    return hits
end
```

## 命中反馈集成

ProjectileSystem.Update() 返回 hitResults 列表，主循环据此触发：
- 环境命中 → 粒子碎片 + 相机微震
- 敌人命中 → 命中停顿 + 方向冲击 + 伤害数字 + 敌人闪白
